// 1. Bubble Sort in Ascending Order with Pass Display
void exercise1_bubbleAscendingPasses() {
    cout << "\n--- Exercise 1: Bubble Sort Ascending (Pass Display) ---\n";
    vector<int> arr = {64, 34, 25, 12, 22, 11, 90};
    cout << "Initial Array: ";
    for(int x : arr) cout << x << " ";
    cout << endl;

    int n = arr.size();
    for(int i = 0; i < n - 1; i++) {
        bool swapped = false;
        for(int j = 0; j < n - i - 1; j++) {
            if(arr[j] > arr[j + 1]) {
                swap(arr[j], arr[j + 1]);
                swapped = true;
            }
        }
        cout << "Pass " << i + 1 << ": ";
        for(int x : arr) cout << x << " ";
        cout << endl;
        if(!swapped) break;
    }
}

// 3. Insertion Sort in Ascending Order with Pass Display
void exercise3_insertionAscendingPasses() {
    cout << "\n--- Exercise 3: Insertion Sort Ascending (Pass Display) ---\n";
    vector<int> arr = {12, 11, 13, 5, 6};
    cout << "Initial Array: ";
    for(int x : arr) cout << x << " ";
    cout << endl;

    int n = arr.size();
    for(int i = 1; i < n; i++) {
        int key = arr[i];
        int j = i - 1;

// Merge Sort Helpers
void merge(vector<int>& arr, int left, int mid, int right, long long& comps) {
    int n1 = mid - left + 1;
    int n2 = right - mid;

    vector<int> L(n1), R(n2);
    for(int i = 0; i < n1; i++) L[i] = arr[left + i];
    for(int j = 0; j < n2; j++) R[j] = arr[mid + 1 + j];

    int i = 0, j = 0, k = left;
    while(i < n1 && j < n2) {
        comps++;
        if(L[i] <= R[j]) {
            arr[k++] = L[i++];
        } else {
            arr[k++] = R[j++];
        }
    }
    while(i < n1) arr[k++] = L[i++];
    while(j < n2) arr[k++] = R[j++];
}

void mergeSortHelper(vector<int>& arr, int left, int right, long long& comps) {
    if(left < right) {
        int mid = left + (right - left) / 2;
        mergeSortHelper(arr, left, mid, comps);
        mergeSortHelper(arr, mid + 1, right, comps);
        merge(arr, left, mid, right, comps);
    }
}

// Quick Sort Helpers
int partitionLast(vector<int>& arr, int low, int high, long long& comps, long long& swaps) {
    int pivot = arr[high];
    int i = (low - 1);
    for(int j = low; j <= high - 1; j++) {
        comps++;
        if(arr[j] < pivot) {
            i++;
            swap(arr[i], arr[j]);
            swaps++;
        }
    }
    swap(arr[i + 1], arr[high]);
    swaps++;
    return (i + 1);
}

void quickSortHelper(vector<int>& arr, int low, int high, long long& comps, long long& swaps, bool displayPass) {
    if(low < high) {
        int pi = partitionLast(arr, low, high, comps, swaps);
        if(displayPass) {
            cout << "Partitioned around pivot " << arr[pi] << ": ";
            for(int x : arr) cout << x << " ";
            cout << endl;
        }
        quickSortHelper(arr, low, pi - 1, comps, swaps, displayPass);
        quickSortHelper(arr, pi + 1, high, comps, swaps, displayPass);
    }
}

// 8. Implement All Five Sorting Algorithms
void exercise8_allFiveSortingAlgorithms() {
    cout << "\n--- Exercise 8: Demonstration of All Five Sorting Algorithms ---\n";
    vector<int> base = {45, 12, 89, 33, 21, 50, 4};

    // 1. Bubble
    vector<int> a1 = base;
    int n = a1.size();
    for(int i = 0; i < n - 1; i++)
        for(int j = 0; j < n - i - 1; j++)
            if(a1[j] > a1[j+1]) swap(a1[j], a1[j+1]);
    cout << "1. Bubble Sort:    "; for(int x : a1) cout << x << " "; cout << endl;

    // 2. Insertion
    vector<int> a2 = base;
    for(int i = 1; i < n; i++) {
        int key = a2[i], j = i - 1;
        while(j >= 0 && a2[j] > key) { a2[j+1] = a2[j]; j--; }
        a2[j+1] = key;
    }
    cout << "2. Insertion Sort: "; for(int x : a2) cout << x << " "; cout << endl;

    // 3. Selection
    vector<int> a3 = base;
    for(int i = 0; i < n - 1; i++) {
        int m = i;
        for(int j = i + 1; j < n; j++) if(a3[j] < a3[m]) m = j;
        swap(a3[i], a3[m]);
    }
    cout << "3. Selection Sort: "; for(int x : a3) cout << x << " "; cout << endl;

    // 4. Merge
    vector<int> a4 = base;
    long long c = 0;
    mergeSortHelper(a4, 0, a4.size() - 1, c);
    cout << "4. Merge Sort:     "; for(int x : a4) cout << x << " "; cout << endl;

    // 5. Quick
    vector<int> a5 = base;
    long long co = 0, sw = 0;
    quickSortHelper(a5, 0, a5.size() - 1, co, sw, false);
    cout << "5. Quick Sort:     "; for(int x : a5) cout << x << " "; cout << endl;
}

// 9. Compare Bubble, Insertion, and Selection (Comparisons & Swaps)
void exercise9_compareBasicSortsMetrics() {
    cout << "\n--- Exercise 9: Metric Comparison of Basic Sorts ---\n";
    vector<int> sample = {30, 10, 50, 20, 40, 60, 5};
    int n = sample.size();

    // Bubble Sort Metrics
    vector<int> t1 = sample;
    long long bComps = 0, bSwaps = 0;
    for(int i = 0; i < n - 1; i++) {
        for(int j = 0; j < n - i - 1; j++) {
            bComps++;
            if(t1[j] > t1[j+1]) {
                swap(t1[j], t1[j+1]);
                bSwaps++;
            }
        }
    }

    // Insertion Sort Metrics
    vector<int> t2 = sample;
    long long iComps = 0, iSwaps = 0;
    for(int i = 1; i < n; i++) {
        int key = t2[i];
        int j = i - 1;
        while(j >= 0) {
            iComps++;
            if(t2[j] > key) {
                t2[j+1] = t2[j];
                iSwaps++;
                j--;
            } else {
                break;
            }
        }
        t2[j+1] = key;
    }

    // Selection Sort Metrics
    vector<int> t3 = sample;
    long long sComps = 0, sSwaps = 0;
    for(int i = 0; i < n - 1; i++) {
        int m = i;
        for(int j = i + 1; j < n; j++) {
            sComps++;
            if(t3[j] < t3[m]) m = j;
        }
        if(m != i) {
            swap(t3[i], t3[m]);
            sSwaps++;
        }
    }

    cout << "Input Array: {30, 10, 50, 20, 40, 60, 5}\n";
    cout << "Algorithm      | Comparisons | Swaps/Shifts\n";
    cout << "-------------------------------------------\n";
    cout << "Bubble Sort    | " << bComps << "         | " << bSwaps << "\n";
    cout << "Insertion Sort | " << iComps << "         | " << iSwaps << "\n";
    cout << "Selection Sort | " << sComps << "         | " << sSwaps << "\n";
}

    if(choice == 1) {

        long long c = 0;
        mergeSortHelper(temp, 0, n - 1, c);

    } else {
        cout << "Invalid choice!\n";
        return;
    }
    for(int x : temp) cout << x << " ";
    cout << endl;

// 12. Sort Array in Ascending and then Descending Order
void exercise12_ascendingThenDescending() {
    cout << "\n--- Exercise 12: Ascending then Descending Order ---\n";
    vector<int> arr = {19, 3, 56, 22, 8, 41};
    cout << "Original Array: ";
    for(int x : arr) cout << x << " ";
    cout << endl;

    // Sort Ascending (using Insertion sort logic)
    int n = arr.size();
    for(int i = 1; i < n; i++) {
        int key = arr[i], j = i - 1;
        while(j >= 0 && arr[j] > key) { arr[j+1] = arr[j]; j--; }
        arr[j+1] = key;
    }
    cout << "Ascending Sorted:  ";
    for(int x : arr) cout << x << " ";
    cout << endl;

    // Sort Descending (reversing or sorting descending)
    for(int i = 0; i < n - 1; i++) {
        for(int j = 0; j < n - i - 1; j++) {
            if(arr[j] < arr[j+1]) swap(arr[j], arr[j+1]);
        }
    }
    cout << "Descending Sorted: ";
    for(int x : arr) cout << x << " ";
    cout << endl;
}

// 14. Demonstrate Best-Case Behavior (Sorted Array)
void exercise14_bestCaseBehavior() {
    cout << "\n--- Exercise 14: Best-Case Behavior (Sorted Array) ---\n";
    vector<int> sortedArr = {1, 2, 3, 4, 5, 6, 7};
    int n = sortedArr.size();

    // Optimized Bubble Sort Best Case
    long long bComps = 0;
    bool swapped = false;
    for(int i = 0; i < n - 1; i++) {
        swapped = false;
        for(int j = 0; j < n - i - 1; j++) {
            bComps++;
            if(sortedArr[j] > sortedArr[j+1]) swapped = true;
        }
        if(!swapped) break; // Terminated in 1 pass
    }

    // Insertion Sort Best Case
    long long iComps = 0;
    for(int i = 1; i < n; i++) {
        iComps++; // 1 comparison per element in sorted array
        int key = sortedArr[i];
        int j = i - 1;
        // loop condition fails immediately
    }

    cout << "Input is already sorted: {1, 2, 3, 4, 5, 6, 7}\n";
    cout << "Optimized Bubble Sort Comparisons: " << bComps << " (O(n))\n";
    cout << "Insertion Sort Comparisons:        " << iComps << " (O(n))\n";
}
